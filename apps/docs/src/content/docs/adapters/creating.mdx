4. ### Define the CMS Types

   ```typescript
   // src/types.ts
   export interface MyCMSConfig {
     apiKey: string;
     projectId: string;
     endpoint?: string;
   }

   export interface MyCMSPage {
     id: string;
     attributes: {
       title: string;
       slug: string;
       publishedAt?: string;
       blocks: MyCMSBlock[];
       seo?: {
         metaTitle?: string;
         metaDescription?: string;
         ogImage?: { url: string };
       };
     };
   }

   export interface MyCMSBlock {
     __type: string;
     id: string;
     [key: string]: any;
   }
   ```

5. ### Implement the Adapter

   ```typescript
   // src/index.ts
   import { IntiAdapter } from '@intijs/core/adapters';
   import type { IntiPage, IntiBlock, IntiNavigation } from '@intijs/core';
   import type { MyCMSConfig, MyCMSPage } from './types.js';

   export class MyCMSAdapter extends IntiAdapter<MyCMSPage> {
     constructor(config: MyCMSConfig) {
       super('MyCMS', config);
     }

     parsePage(raw: MyCMSPage): IntiPage {
       const { id, attributes } = raw;
       
       return {
         id,
         slug: attributes.slug,
         title: attributes.title,
         blocks: this.parseBlocks(attributes.blocks),
         seo: attributes.seo ? {
           title: attributes.seo.metaTitle,
           description: attributes.seo.metaDescription,
           image: attributes.seo.ogImage?.url
         } : undefined
       };
     }

     private parseBlocks(blocks: MyCMSBlock[]): IntiBlock[] {
       return blocks.map(block => ({
         type: block.__type,
         id: block.id,
         data: Object.fromEntries(
           Object.entries(block).filter(([key]) => 
             !['__type', 'id'].includes(key)
           )
         )
       }));
     }

     async introspect(): Promise<SchemaDefinition> {
       // Optional but recommended
       const response = await fetch(
         `${this.config.endpoint}/schema`,
         { headers: { 'Authorization': `Bearer ${this.config.apiKey}` } }
       );
       
       const cmsSchema = await response.json();
       
       return {
         source: 'MyCMS',
         version: '1.0.0',
         collections: cmsSchema.contentTypes.map((ct: any) => ({
           name: ct.name,
           description: ct.description,
           fields: ct.attributes.map((attr: any) => ({
             name: attr.name,
             type: this.mapFieldType(attr.type),
             required: attr.required,
             description: attr.description
           }))
         }))
       };
     }

     private mapFieldType(cmsType: string): FieldType {
       const mapping: Record<string, FieldType> = {
         'string': 'string',
         'text': 'string',
         'number': 'number',
         'boolean': 'boolean',
         'datetime': 'date',
         'json': 'json',
         'media': 'media',
         'relation': 'relation'
       };
       return mapping[cmsType] || 'string';
     }
   }

   export default MyCMSAdapter;
   ```

## Testing the Adapter

### Unit Test

```typescript
// tests/adapter.test.ts
import { describe, it, expect } from 'bun:test';
import { MyCMSAdapter } from '../src/index';

describe('MyCMSAdapter', () => {
  const adapter = new MyCMSAdapter({
    apiKey: 'test',
    projectId: 'test'
  });

  it('should parse page correctly', () => {
    const raw = {
      id: '123',
      attributes: {
        title: 'Test Page',
        slug: 'test-page',
        blocks: [
          { __type: 'hero', id: 'b1', title: 'Hero Title' }
        ]
      }
    };

    const page = adapter.parsePage(raw);

    expect(page.id).toBe('123');
    expect(page.title).toBe('Test Page');
    expect(page.blocks).toHaveLength(1);
    expect(page.blocks[0].type).toBe('hero');
  });

  it('should handle missing blocks gracefully', () => {
    const raw = {
      id: '456',
      attributes: {
        title: 'Empty Page',
        slug: 'empty',
        blocks: []
      }
    };

    const page = adapter.parsePage(raw);
    expect(page.blocks).toEqual([]);
  });
});
```

### Integration Test

```typescript
// tests/integration.test.ts
import { describe, it, expect } from 'bun:test';
import { MyCMSAdapter } from '../src/index';

describe('MyCMSAdapter Integration', () => {
  it('should introspect schema', async () => {
    const adapter = new MyCMSAdapter({
      apiKey: process.env.MYCMS_API_KEY!,
      projectId: process.env.MYCMS_PROJECT_ID!,
      endpoint: process.env.MYCMS_ENDPOINT
    });

    const schema = await adapter.introspect();

    expect(schema.source).toBe('MyCMS');
    expect(schema.collections).toBeDefined();
    expect(schema.collections.length).toBeGreaterThan(0);
  });
});
```
